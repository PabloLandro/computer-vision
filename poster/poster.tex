\begin{column}{0.22\paperwidth}

  \begin{block}{Introduction}
    \begin{itemize}
      \item Goal: extract \textbf{recipe steps} from egocentric cooking videos.
      \item Challenge: cluttered kitchens, occlusions, and background actions.
      \item Backbone: frozen \textbf{V-JEPA2} video embeddings.
      \item Classifier: lightweight temporal model on top of embeddings.
    \end{itemize}

    \vskip0.3em
    \textbf{Two-stage pipeline}
    \begin{enumerate}
      \item \textbf{Relevance filter}: cooking vs. background.
      \item \textbf{Action classifier}: verb, noun, and verb--noun pair.
    \end{enumerate}
  \end{block}

  \begin{block}{Dataset: EPIC-KITCHENS-100}
    \begin{itemize}
      \item \textbf{408} embedded videos.
      \item \textbf{170,299} temporal blocks.
      \item \textbf{1024-D} V-JEPA2 embedding per 64-frame block.
      \item \textbf{116,895} blocks overlap an annotation.
      \item \textbf{53,404} background/unannotated blocks.
      \item \textbf{7,008} dataset action pairs;
            \textbf{2,721} action-head classes.
    \end{itemize}

    \vskip0.3em
    \begin{center}
    \begin{tabular}{lr}
      \toprule
      Block label & Fraction \\
      \midrule
      Relevant    & 27.9\% \\
      Non-relevant & 72.1\% \\
      \bottomrule
    \end{tabular}
    \end{center}

  \end{block}

  \begin{block}{Why V-JEPA2-small?}
    \begin{itemize}
      \item \textbf{Small} V-JEPA2 variant.
      \item Frozen backbone: no video-model fine-tuning.
      \item Cloud compute for dataset embedding.
      \item Classifier training on Apple MPS.
    \end{itemize}

    \vskip0.3em
    \begin{center}

  \begin{block}{Classification Heads}
    \begin{columns}[t]
      \begin{column}{0.48\linewidth}
        \textbf{Temporal encoder}
        \begin{itemize}
          \item \textbf{7-block window} around target.
          \item Zero-padding at boundaries.
          \item 1024-D embeddings $\rightarrow$ \textbf{768-D}.
          \item Learned position embeddings.
          \item \textbf{2-layer, 8-head TransformerEncoder}.
        \end{itemize}
      \end{column}
      \begin{column}{0.48\linewidth}
        \textbf{Heads and decoding}
        \begin{itemize}
          \item Center token feeds all heads.
          \item Heads: relevance, verb, noun, action pair.
          \item Relevance trains on \textbf{all} blocks.
          \item Action losses use annotated relevant blocks.
          \item Decoding adds pair prior, $w=0.25$.
          \item Single-model and 3-model ensemble with averaged predictions.
        \end{itemize}
      \end{column}
    \end{columns}
  \end{block}

% ---- RIGHT COLUMN ----
\begin{column}{0.22\paperwidth}

  \begin{block}{Why a Transformer Encoder?}
    \begin{itemize}
      \item Single blocks miss before/after context.
      \item Attention compares all 7 neighboring blocks.
      \item The model can see action setup and completion.
      \item Masking handles short windows at video boundaries.
      \item Only the center block is classified.
    \end{itemize}
  \end{block}

  \begin{block}{Results}
    \begin{center}
    \begin{tabular}{lcc}
      \toprule
      Metric & Val & Test \\
      \midrule
      Relevance accuracy & 91.6\% & 92.3\% \\
      Relevance F1       & 84.8\% & 86.1\% \\
      Verb accuracy      & 94.4\% & 95.6\% \\
      Noun accuracy      & 98.3\% & 98.7\% \\
      Exact action       & 93.8\% & 95.3\% \\
      \bottomrule
    \end{tabular}
    \end{center}

    \vskip0.3em
    \begin{itemize}
      \item Action rows: predicted relevant blocks for the ensemble.
      \item Oracle relevance: \textbf{91.8\%} test exact action.
      \item Predicted relevant blocks: \textbf{95.3\%} test exact action.
    \end{itemize}
  \end{block}

  \begin{block}{LLM Post-Processing}
    \begin{itemize}
      \item Input: sequence of predicted \textbf{(verb, noun)} pairs.
      \item Model: \textbf{Llama 3.2}.
      \item Output: readable recipe steps.
      \item Repeated events are merged.
      \item Sparse predictions become natural instructions.
    \end{itemize}
  \end{block}

  \begin{block}{Conclusion}
    \begin{itemize}
      \item Frozen V-JEPA2 embeddings reduce training cost.
      \item Relevance filtering removes background noise.
      \item Transformer context improves block-level decisions.
      \item Multi-head decoding predicts coherent verb--noun actions.
      \item LLM post-processing turns actions into recipes.
    \end{itemize}
  \end{block}

\end{column}
